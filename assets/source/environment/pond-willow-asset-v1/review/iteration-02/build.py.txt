class Geo:
 def __init__(self,name):self.name=name;self.v=[];self.f=[];self.c=[]
 def add(self,v,f,c):
  o=len(self.v);self.v.extend(v);self.f.extend([tuple(o+i for i in f0) for f0 in f]);self.c.extend(c if isinstance(c,list) else [c]*len(f))
 def tube(self,pts,radii,n=7):
  pts=[Vector(p) for p in pts];v=[]
  for k,p in enumerate(pts):
   d=pts[min(k+1,len(pts)-1)]-pts[max(0,k-1)];q=d.to_track_quat('Z','Y')
   for j in range(n):v.append(tuple(p+q@Vector((radii[k]*math.cos(j*math.tau/n),radii[k]*math.sin(j*math.tau/n),0))))
  f=[tuple(reversed(range(n))),tuple((len(pts)-1)*n+j for j in range(n))];c=[0,1]
  for k in range(len(pts)-1):
   for j in range(n):f.append((k*n+j,k*n+(j+1)%n,(k+1)*n+(j+1)%n,(k+1)*n+j));c.append([1,2,1,0,1,2,3][j%7])
  self.add(v,f,c)
 def obj(self):
  me=bpy.data.meshes.new(self.name+'_Mesh');me.from_pydata(self.v,[],self.f);me.update();me.materials.append(mat)
  uv=me.uv_layers.new(name='PaletteUV')
  for p,c in zip(me.polygons,self.c):
   for i in p.loop_indices:uv.data[i].uv=((c+.5)/len(PAL),.5)
  bm=bmesh.new();bm.from_mesh(me);bmesh.ops.recalc_face_normals(bm,faces=list(bm.faces));bm.to_mesh(me);bm.free()
  ob=bpy.data.objects.new(self.name,me);bpy.context.collection.objects.link(ob);return ob
